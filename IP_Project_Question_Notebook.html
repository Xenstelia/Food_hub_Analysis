# Write the code here
# Filter restaurants based on rating criteria.
foodhub_df_copy = foodhub_df.copy()
foodhub_df_copy['rating'] = foodhub_df_copy['rating'].replace('Not given', np.nan)
foodhub_df_copy['rating'] = pd.to_numeric(foodhub_df_copy['rating'])

# Calculate mean rating for each restaurant.
restaurant_ratings = foodhub_df_copy.groupby('restaurant_name')['rating'].agg(['mean', 'count'])

# Filter restaurants meeting criteria.
eligible_restaurants = restaurant_ratings[(restaurant_ratings['count'] > 50) & (restaurant_ratings['mean'] > 4)]

# Display the results.
print("\nRestaurants meeting promotional offer criteria:")
print(eligible_restaurants)
